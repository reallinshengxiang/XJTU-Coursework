1、数据库事务的练习
（1）、在 employees 表中，为员工 emp_no 为 10003 的员工新增一条工作经历记录到 dept_emp表，同时在 titles 表中为其添加一个新的职位记录。请编写一个事务，确保这两个操作要么都成功提交，要么都不生效，并通过 SELECT 给出事务提交前后查询相关数据的 SQL 语句及预期结果。
-- 1. 事务提交前查询原始数据
SELECT 'Before Transaction (dept_emp)' AS state, * FROM kk.dept_emp WHERE emp_no = 10003;
SELECT 'Before Transaction (titles)' AS state, * FROM kk.titles WHERE emp_no = 10003;
-- 2. 执行事务（包含插入操作）
START TRANSACTION;
INSERT INTO kk.dept_emp (emp_no, dept_no, from_date, to_date)
VALUES (10003, 'd001', '2023-01-01', '2024-01-01');
INSERT INTO kk.titles (emp_no, title, from_date, to_date)
VALUES (10003, 'Market Salesman', '2023-01-01', '2024-01-01');
COMMIT;
-- 3. 事务提交后查询原始数据
SELECT 'After Transaction (dept_emp)' AS state, * FROM kk.dept_emp WHERE emp_no = 10003;
SELECT 'After Transaction (titles)' AS state, * FROM kk.titles WHERE emp_no = 10003;

（2）需要完成一个事务，要对 salaries 表中 emp_no 为 10001 的员工薪资进行调整，先将薪资提高 20%，然后模拟一个错误操作（如将 to_date 设置为一个不合理的过去时间），使用事务回滚确保数据库数据不受错误操作影响。请编写完整的事务代码，并给出事务回滚前后查询该员工薪资数据的 SQL 语句及预期结果。
-- 事务回滚前查询原始数据
SELECT 'Before Transaction' AS state, * FROM kk.salaries 
WHERE emp_no = 10001 
ORDER BY from_date DESC
LIMIT 10;
-- 开始事务
START TRANSACTION;
-- 步骤1：获取当前薪资并提高20%
UPDATE kk.salaries 
SET to_date = CURRENT_DATE - 1
WHERE emp_no = 10001 AND to_date = '9999-01-01';
INSERT INTO kk.salaries (emp_no, salary, from_date, to_date)
SELECT 
    emp_no, 
    CAST(salary * 1.2 AS INT),  -- 提高20%
    CURRENT_DATE, 
    '9999-01-01'
FROM kk.salaries
WHERE emp_no = 10001 AND to_date = CURRENT_DATE - 1;
-- 事务进行中查询原始数据
SELECT 'On Transaction' AS state, * FROM kk.salaries 
WHERE emp_no = 10001 
ORDER BY from_date DESC
LIMIT 10;
-- 步骤2：模拟错误操作（设置不合理的过去时间）
-- 尝试插入一条to_date早于from_date的记录
INSERT INTO kk.salaries (emp_no, salary, from_date, to_date)
VALUES (10001, 10000, '2025-05-31', '2020-01-01');  -- to_date < from_date
-- 显式回滚（实际不会执行到这里，因为上条INSERT会失败）
ROLLBACK;
-- 事务回滚后查询数据
SELECT 'After Transaction (Rollback)' AS state, * FROM kk.salaries 
WHERE emp_no = 10001 
ORDER BY from_date DESC
LIMIT 10;

（3）在 departments 表中，要对部门信息进行一系列操作。首先插入一个新部门记录，设置保存点；接着尝试修改一个不存在的部门编号（模拟错误操作）；然后回滚到保存点，再插入另一个新部门记录，最后提交事务。请编写完整的事务代码，并给出各个关键步骤（设置保存点、回滚到保存点、提交事务）前后查询 departments 表数据的 SQL 语句及预期结果。
-- 事务开始前查询原始数据
SELECT 'Before Transaction' AS state, * FROM kk.departments ORDER BY dept_no;
-- 开始事务
START TRANSACTION;
-- 步骤1：插入第一个新部门
INSERT INTO kk.departments (dept_no, dept_name)
VALUES ('d010', 'Learning');
-- 设置保存点
SAVEPOINT sp1;
-- 步骤2：尝试修改不存在的部门（模拟错误操作）
UPDATE kk.departments 
SET dept_name = 'Invalid Department' 
WHERE dept_no = 'd999';  -- 不存在的部门编号
-- 保存点后查询（显示第一个部门已插入）
SELECT 'After Savepoint (before rollback)' AS state, * 
FROM kk.departments 
WHERE dept_no IN ('d010', 'd011')
ORDER BY dept_no;
SELECT 'After Savepoint (before rollback)' AS state, * FROM kk.departments ORDER BY dept_no;
-- 步骤3：回滚到保存点（撤销错误操作）
ROLLBACK TO SAVEPOINT sp1;
-- 回滚后查询（错误操作已撤销）
SELECT 'After Rollback to Savepoint' AS state, * FROM kk.departments ORDER BY dept_no;
-- 步骤4：插入第二个新部门
INSERT INTO kk.departments (dept_no, dept_name)
VALUES ('d011', 'Policy');
-- 提交前查询（两个部门都存在）
SELECT 'Before Commit' AS state, *  FROM kk.departments ORDER BY dept_no;
-- 提交事务
COMMIT;
-- 事务结束后查询最终数据
SELECT 'After Commit' AS state, *  FROM kk.departments ORDER BY dept_no;

\c employ test_user1
-- 开始事务并修改数据
BEGIN;
UPDATE kk.concurrency_test SET value = 'Dirty Read Attempt' WHERE id = 1;
-- 保持事务开放，不提交
-- 连接到 employ 数据库
\c employ test_user2
-- 设置隔离级别为 READ COMMITTED
SET TRANSACTION ISOLATION LEVEL READ COMMITTED;

b)
